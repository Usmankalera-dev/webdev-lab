This is the final project of the course, and it brings the whole JavaScript module together: the DOM, events, arrays, objects and localStorage, all in one working app. You built the pieces in Lesson 18 — now you assemble them into a polished page with styling, a counter, and a clear-completed button. When this project works, you can honestly say you built a real web app.

Step 1 — The HTML shell

The app needs an input, an Add button, a counter, a list, and a Clear completed button:

<div class="todo-app">
  <h2>My Tasks</h2>
  <input id="taskInput" placeholder="New task...">
  <button id="addBtn">Add</button>
  <p id="counter">0 tasks left</p>
  <ul id="taskList"></ul>
  <button id="clearBtn">Clear completed</button>
</div>

Step 2 — Style the app

Give the app a card look, and make done tasks show a line through them — the visual signal that a task is finished:

.todo-app {
  max-width: 480px;
  margin: 0 auto;
  background: #ffffff;
  padding: 24px;
  border-radius: 16px;
  box-shadow: 0 4px 16px rgba(0, 0, 0, 0.1);
}
#taskList { list-style: none; padding: 0; }
#taskList li {
  padding: 10px;
  border-bottom: 1px solid #e5e7eb;
  cursor: pointer;
  display: flex;
  justify-content: space-between;
  align-items: center;
}
#taskList li.done { text-decoration: line-through; color: #9ca3af; }

Step 3 — Add and render tasks

The JavaScript is the same pattern as Lesson 18: an array of task objects, a render() function that rebuilds the list, and an Add handler that pushes a new task:

let tasks = [];
let input = document.getElementById("taskInput");
let list = document.getElementById("taskList");
let counter = document.getElementById("counter");

function render() {
  list.innerHTML = "";
  let left = 0;
  tasks.forEach(function (task, index) {
    if (!task.done) left++;
    let li = document.createElement("li");
    li.textContent = task.text;
    if (task.done) li.classList.add("done");
    li.addEventListener("click", function () {
      task.done = !task.done;
      saveAndRender();
    });
    let del = document.createElement("button");
    del.textContent = "Delete";
    del.addEventListener("click", function (e) {
      e.stopPropagation();
      tasks.splice(index, 1);
      saveAndRender();
    });
    li.appendChild(del);
    list.appendChild(li);
  });
  counter.textContent = left + " task" + (left === 1 ? "" : "s") + " left";
}

document.getElementById("addBtn").addEventListener("click", function () {
  let text = input.value.trim();
  if (text === "") return;
  tasks.push({ text: text, done: false });
  input.value = "";
  saveAndRender();
});

The counter counts only tasks that are not done, and the word "task" becomes "tasks" when the count is not 1 — a small detail that makes the app feel finished.

Step 4 — Clear completed and save

The Clear completed button keeps only the tasks that are not done, using filter — a method that returns a new array with only the items that pass a test:

document.getElementById("clearBtn").addEventListener("click", function () {
  tasks = tasks.filter(function (task) {
    return !task.done;
  });
  saveAndRender();
});

function saveAndRender() {
  localStorage.setItem("todo-tasks", JSON.stringify(tasks));
  render();
}

let saved = localStorage.getItem("todo-tasks");
if (saved) tasks = JSON.parse(saved);
render();

filter is worth learning properly: it takes a test function, runs it on every item, and keeps only the ones where the test returns true. Here the test is !task.done — keep the task if it is not done.

Step 5 — Test the whole app

Open the page and run through the full test: add three tasks, click one to mark it done, delete another, click Clear completed, and check the counter. Then refresh the page — your remaining tasks must still be there, because of localStorage. If everything works, you have built a complete web application. Add it to your portfolio: it is the strongest proof of what this course taught you.
